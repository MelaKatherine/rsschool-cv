## 🧩 Code Sample

```
async function loadMountains() {
  try {
    const response = await fetch('https://api.example.com/mountains');
    const data = await response.json();
    const list = document.getElementById('mountain-list');
    list.innerHTML = '';

    data.forEach((mountain) => {
      const item = document.createElement('li');
      item.textContent = `${mountain.name} – ${mountain.height} m`;
      list.appendChild(item);
    });
  } catch (error) {
    console.error('Error loading mountains:', error);
  }
}

document.addEventListener('DOMContentLoaded', loadMountains);
```
